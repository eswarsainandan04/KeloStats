WITH
cuisine_pop AS (
  SELECT r.cuisine, COUNT(o.order_id) AS orders_count
  FROM orders o
  JOIN restaurants r ON o.restaurant_id = r.restaurant_id
  GROUP BY r.cuisine
),
restaurant_pop AS (
  SELECT o.restaurant_id, COUNT(o.order_id) AS orders_count
  FROM orders o
  GROUP BY o.restaurant_id
),
rating_stats AS (
  SELECT restaurant_id, rating
  FROM restaurants
),
menu_price_stats AS (
  SELECT r.cuisine, AVG(m.price) AS avg_price
  FROM menu_items m
  JOIN restaurants r ON m.restaurant_id = r.restaurant_id
  GROUP BY r.cuisine
),
customer_dist AS (
  SELECT city, COUNT(customer_id) AS customer_count
  FROM customers
  GROUP BY city
),
signup_trends AS (
  SELECT date_trunc('month', signup_date) AS month, COUNT(*) AS new_customers
  FROM customers
  GROUP BY month
  ORDER BY month
),
orders_over_time AS (
  SELECT date_trunc('day', order_time) AS day, COUNT(*) AS orders
  FROM orders
  GROUP BY day
  ORDER BY day
),
order_status_dist AS (
  SELECT status, COUNT(*) AS cnt
  FROM orders
  GROUP BY status
),
delivery_time_stats AS (
  SELECT AVG(EXTRACT(EPOCH FROM (delivery_time - order_time))/60) AS avg_delivery_minutes
  FROM orders
  WHERE delivery_time IS NOT NULL
),
quantity_stats AS (
  SELECT item_id, AVG(quantity)::numeric(10,2) AS avg_quantity
  FROM order_items
  GROUP BY item_id
),
order_value_stats AS (
  SELECT o.order_id, SUM(oi.quantity * oi.price) AS order_total
  FROM orders o
  JOIN order_items oi ON o.order_id = oi.order_id
  GROUP BY o.order_id
),
customer_order_freq AS (
  SELECT customer_id, COUNT(*) AS order_count
  FROM orders
  GROUP BY customer_id
)
SELECT jsonb_build_object(
